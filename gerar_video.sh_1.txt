#!/usr/bin/env bash
# gerar_video.sh <url_do_app> <slug>
# Usa o video aprovado (base.mp4) e troca SO a marca d'agua pela logo da barbearia.
# - fila automatica: se dois pedidos chegarem juntos, o segundo espera o primeiro
#   (na VPS de 1 vCPU, dois ao mesmo tempo travavam os dois)
# - gera tambem a capa (<slug>_capa.jpg) para usar como previa
set -e
URL="$1"; SLUG="$2"
[ -z "$SLUG" ] && { echo "uso: bash gerar_video.sh <url_do_app> <slug>"; exit 1; }
BASE="$(cd "$(dirname "$0")" && pwd)"
[ -f "$BASE/base.mp4" ] || { echo "ERRO: falta base.mp4 no repo"; exit 1; }
W="$BASE/work/$SLUG"; mkdir -p "$W" "$BASE/saidas"

# ── fila: um vídeo de cada vez (espera até 25 min pela vez) ──
exec 9>"$BASE/.fila.lock"
if ! flock -w 1500 9; then echo "ERRO: fila cheia, tente de novo"; exit 1; fi

echo "==> 1/4 capturando a logo do app..."
node "$BASE/logo_cliente.js" "$URL" "$W"
echo "==> 2/4 montando a marca d'agua..."
python3 "$BASE/marca.py" "$W/splash.png" "$W/marca.png"
echo "==> 3/4 aplicando no video..."
# WhatsApp recusa video acima de ~16 MB. Alvo: ~10 MB, mantendo 1080x1920.
ffmpeg -y -loglevel error -i "$BASE/base.mp4" -i "$W/marca.png" \
  -filter_complex "[0:v][1:v]overlay=x=(main_w-overlay_w)/2:y=1440:format=auto[v]" \
  -map "[v]" -map 0:a -c:v libx264 -preset veryfast -crf 28 \
  -maxrate 950k -bufsize 1900k -pix_fmt yuv420p \
  -c:a aac -b:a 128k -movflags +faststart "$BASE/saidas/$SLUG.mp4"
echo "==> 4/4 gerando a capa..."
ffmpeg -y -loglevel error -ss 84 -i "$BASE/saidas/$SLUG.mp4" -frames:v 1 -q:v 3 \
  "$BASE/saidas/${SLUG}_capa.jpg"
echo "=================================================="
echo "  VIDEO PRONTO: saidas/$SLUG.mp4  ($(du -m "$BASE/saidas/$SLUG.mp4" | cut -f1) MB)"
echo "  CAPA:         saidas/${SLUG}_capa.jpg"
echo "=================================================="
